// DOM elements
const selectedPreview = document.getElementById('selectedIconPreview');
const iconPickerModal = document.getElementById('iconPickerModal');
const modalOverlay = document.getElementById('modalOverlay');
const iconGrid = document.getElementById('iconGrid');
const iconSearch = document.getElementById('iconSearch');
const closePicker = document.getElementById('closePicker');
const selectedIconInput = document.getElementById('selectedIcon');

// Load icons into grid
function loadIcons(icons = fontAwesomeIcons) {
    iconGrid.innerHTML = '';
    icons.forEach(icon => {
        const iconDiv = document.createElement('div');
        iconDiv.className = 'icon-item';
        iconDiv.setAttribute('data-icon', icon.name);
        iconDiv.innerHTML = `
            <i class="fas ${icon.name}"></i>
            <span>${icon.label}</span>
        `;
        
        iconDiv.addEventListener('click', () => {
            selectIcon(icon.name, icon.label);
        });
        
        iconGrid.appendChild(iconDiv);
    });
}

// Select icon
function selectIcon(iconName, iconLabel) {
    // Update preview
    selectedPreview.innerHTML = `<i class="fas ${iconName}"></i><span>${iconLabel}</span>`;
    
    // Update hidden input
    selectedIconInput.value = iconName;
    
    // Close modal
    closeIconPicker();
    
    // Clear error
    document.getElementById('icon-error').textContent = '';
    
    // Highlight selected icon in grid (if modal is reopened)
    document.querySelectorAll('.icon-item').forEach(item => {
        if (item.getAttribute('data-icon') === iconName) {
            item.classList.add('selected');
        } else {
            item.classList.remove('selected');
        }
    });
}

// Open icon picker
function openIconPicker() {
    iconPickerModal.style.display = 'block';
    modalOverlay.style.display = 'block';
    loadIcons();
}

// Close icon picker
function closeIconPicker() {
    iconPickerModal.style.display = 'none';
    modalOverlay.style.display = 'none';
}

// Search icons
iconSearch.addEventListener('input', (e) => {
    const searchTerm = e.target.value.toLowerCase();
    const filteredIcons = fontAwesomeIcons.filter(icon => 
        icon.name.toLowerCase().includes(searchTerm) || 
        icon.label.toLowerCase().includes(searchTerm)
    );
    loadIcons(filteredIcons);
});

// Event listeners
selectedPreview.addEventListener('click', openIconPicker);
closePicker.addEventListener('click', closeIconPicker);
modalOverlay.addEventListener('click', closeIconPicker);

// Set default icon
selectIcon('fa-seedling', 'Seedling');

// Form validation
function validateIcon() {
    const icon = selectedIconInput.value;
    if (!icon) {
        document.getElementById('icon-error').textContent = 'Please select an icon for the plan';
        return false;
    }
    return true;
}
